  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  // Absolute (document-relative) offsets, recomputed only on load/resize —
  // never inside the scroll handler. Reading getBoundingClientRect() on
  // every scroll frame is what forces the browser to recompute layout
  // repeatedly; caching these once and comparing against window.scrollY
  // (a plain number, no layout read) avoids that entirely.
  let heroBottomAbs = 0;
  let sectionTopsAbs = [];
  function recalcOffsets(){
    const y = window.scrollY;
    heroBottomAbs = heroSection.getBoundingClientRect().bottom + y;
    sectionTopsAbs = sections.map(s => s.getBoundingClientRect().top + y);
  }

  function updateOnScroll(){
    const scrollY = window.scrollY;
    const innerHeight = window.innerHeight;
    const scrollable = document.documentElement.scrollHeight - innerHeight;

    const showBackToTop = scrollY > innerHeight;
    const showStickyBar = scrollY > heroBottomAbs;
    let current = sections[0];
    for(let i = 0; i < sections.length; i++){ if(sectionTopsAbs[i] - 120 <= scrollY){ current = sections[i]; } }
    const frac = scrollable > 0 ? Math.min(1, Math.max(0, scrollY / scrollable)) : 0;

    backToTop.classList.toggle('visible', showBackToTop);
    stickyBar.classList.toggle('show', showStickyBar);
    railLinks.forEach(a => a.classList.toggle('active', a.getAttribute('href') === '#' + current.id));
    if(railFill) railFill.style.height = (frac * 100) + '%';

    scrollTicking = false;
  }

    const startY = window.scrollY;
    const distance = targetY - startY;
    const startTime = performance.now();
    function step(now){
      const elapsed = now - startTime;
      const progress = Math.min(elapsed / duration, 1);
      const eased = 0.5 - Math.cos(progress * Math.PI) / 2;
      window.scrollTo(0, startY + distance * eased);
      if(progress < 1){ requestAnimationFrame(step); }
